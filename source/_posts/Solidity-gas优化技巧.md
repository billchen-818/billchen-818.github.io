## 存储打包

下面两个合约，如下：

未优化的合约：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract PackingBad {
    uint256 public a; // 32 bytes
    uint8 public b;   // 1 byte
    uint256 public c; // 32 bytes
    uint8 public d;   // 1 byte
    uint256 public e; // 32 bytes
    uint8 public f;   // 1 byte

    constructor() {
    
    }

    function setValue() public {
        a = 1;
        b = 2;
        c = 3;
        d = 4;
        e = 5;
        f = 6;
    }
}
```

优化过后的合约：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract PackingGood {
    uint256 public a; // 32 bytes
    uint256 public c; // 32 bytes
    uint256 public e; // 32 bytes
    uint8 public b; // 1 byte
    uint8 public d; // 1 byte
    uint8 public f; // 1 byte

    constructor() {
    }

    function setValues() public {
        a = 1;
        b = 2;
        c = 3;
        d = 4;
        e = 5;
        f = 6;
    }
}
```

## 常量

solidity中，存在`constant`和`immutable`两种常量声明方案，它们的区别如下：

- constant：在solidity编译过程中被写入字节码
- immutable：在solidity合约部署过程中被写入字节码，所以我们可以在构造函数`constructor`内对其进行赋值

需要说明的是，运行时低成本gas对应着部署时高成本gas，但是在部署时多消耗的gas肯定远小于多次调用时消耗的gas。请看源码：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract ConstantBad {
    uint256 public A = 1;

    constructor() {
    }

    function getValues() public view returns (uint256) {
        return A;
    }
}
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract ConstantGood {
    uint256 public constant A = 1;

    constructor() {
    }

    function getValues() public pure returns (uint256) {
        return A;
    }
}
```

## 缓存

缓存可以使用临时变量来避免多次从存储中读取，每次从存储读取消耗2100gas，看代码：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract CacheBad {
    uint256 public count = 20;

    constructor() {
    }

    function add(uint256 input) public view returns (uint256) {
        uint256 res = 0;
        
        for (uint256 i = 0; i < count; i++) {
            // do nothing
            res += input;
        }
        return res;
    }
}
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract CacheGood {
    uint256 public count = 20;

    constructor() {
    }

    function add(uint256 input) public view returns (uint256) {
        uint256 res = 0;
        uint256 aa = count;

        for (uint256 i = 0; i < aa; i++) {
            res += input;
        }
        return res;
    }
}
```

我们在编写两个测试函数，使用`forge test --gas-report`来测试，测试结果如下：

## Calldata与memory

calldata与memory是EVM的两个重要区域，calldata区域存储来自用户或者其它合约的请求参数，内容不可变，可以通过`CALLDATALOAD`操作码把内容读取到栈上，也可以通过`CALLDATACOPY`将内容读取到内存中。其中`CALLDATALOAD`消耗3gas，`CALLDATACOPY`取决于具体长度。

直接看代码：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract CallDataBad {
    string public name;

    function setName(string memory _name) external {
        name = _name;
    }
}
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract CallDataGood {
    string public name;
    
    function setName(string calldata _name) external {
        name = _name;
    }
}
```
